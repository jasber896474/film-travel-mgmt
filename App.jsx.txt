const sb = async (path, opts = {}) => {
  const res = await fetch(`${SUPABASE_URL}/rest/v1/${path}`, {
    headers: {
      apikey: SUPABASE_KEY,
      Authorization: `Bearer ${SUPABASE_KEY}`,
      "Content-Type": "application/json",
      Prefer: opts.prefer || "return=representation",
    },
    ...opts,
  });
  if (!res.ok) {
    const err = await res.text();
    throw new Error(err);
  }
  const text = await res.text();
  return text ? JSON.parse(text) : [];
};

function HotelStayForm({ init, hotels, pricingRules, onSave, onClose }) {
  const blank = { hotel_id: "", room_type: "Single", room_custom: "", check_in: "", check_out: "", base_price: "" };
  const [f, setF] = useState(init ? { ...init, hotel_id: init.hotel_id || "" } : blank);
  const set = k => e => setF(p => ({ ...p, [k]: e.target.value }));
  const nights = diffDays(f.check_in, f.check_out);
  const { totalAmount, breakdown } = useMemo(() => {
    if (!f.check_in || !f.check_out || nights === 0) return { totalAmount: 0, breakdown: [] };
    const roomLabel = f.room_type === "Custom" ? f.room_custom : f.room_type;
    let total = 0; const bd = [];
    for (let i = 0; i < nights; i++) {
      const d = new Date(f.check_in); d.setDate(d.getDate() + i);
      const ds = d.toISOString().slice(0, 10);
      const rule = pricingRules.find(r => +r.hotel_id === +f.hotel_id && r.date === ds &&
        (r.room_type === roomLabel || (r.room_type === "Custom" && r.room_custom === roomLabel)));
      const price = rule ? (+rule.final_price || 0) : (+f.base_price || 0);
      total += price; bd.push({ date: ds, price, fromRule: !!rule });
    }
    return { totalAmount: total, breakdown: bd };
  }, [f, nights, pricingRules]);
  return (
    <div>
      <div className="grid grid-cols-2 gap-3">
        <Field label="ホテル">
          <select className={sel} value={f.hotel_id} onChange={set("hotel_id")}>
            <option value="">-- 選択 --</option>
            {hotels.map(h => <option key={h.id} value={h.id}>{h.name}</option>)}
          </select>
        </Field>
        <Field label="Room Type">
          <select className={sel} value={f.room_type} onChange={set("room_type")}>{ROOM_TYPES.map(r => <option key={r}>{r}</option>)}</select>
        </Field>
        {f.room_type === "Custom" && <Field label="Custom Room Type"><input className={inp} value={f.room_custom} onChange={set("room_custom")} /></Field>}
        <Field label="チェックイン"><input type="date" className={inp} value={f.check_in || ""} onChange={set("check_in")} /></Field>
        <Field label="チェックアウト"><input type="date" className={inp} value={f.check_out || ""} onChange={set("check_out")} /></Field>
        <Field label="基本料金 ($/泊)"><input type="number" className={inp} value={f.base_price} onChange={set("base_price")} /></Field>
        <Field label="宿泊日数"><input className={inp} value={nights || ""} readOnly style={{ background: "#f5f5f5" }} /></Field>
        <Field label="合計金額 ($)"><input className={inp} value={totalAmount ? `$${totalAmount.toLocaleString()}` : ""} readOnly style={{ background: "#f5f5f5", fontWeight: 700, color: "#2563eb" }} /></Field>
      </div>
      {breakdown.length > 0 && (
        <div style={{ marginTop: 10, background: "#f8fafc", borderRadius: 8, padding: 10 }}>
          <div style={{ fontSize: 11, fontWeight: 700, color: "#6b7280", marginBottom: 6 }}>日別料金内訳</div>
          <div style={{ display: "flex", flexWrap: "wrap", gap: 4 }}>
            {breakdown.map(b => (
              <span key={b.date} style={{ fontSize: 11, padding: "2px 7px", borderRadius: 4, background: b.fromRule ? "#dbeafe" : "#f3f4f6", color: b.fromRule ? "#1d4ed8" : "#374151", border: b.fromRule ? "1px solid #93c5fd" : "1px solid #e5e7eb" }}>
                {b.date.slice(5)} ${b.price.toLocaleString()}{b.fromRule ? " *" : ""}
              </span>
            ))}
          </div>
        </div>
      )}
      <div className="flex justify-end gap-2 mt-4">
        <button onClick={onClose} className="px-4 py-2 rounded-lg text-sm border hover:bg-gray-50">キャンセル</button>
        <button onClick={() => onSave({ ...f, nights, total_amount: totalAmount })} style={{ background: "#2563eb", color: "white", border: "none", borderRadius: 8, padding: "8px 18px", fontWeight: 700, cursor: "pointer", fontSize: 13 }}>保存</button>
      </div>
    </div>
  );
}

  const [persons, setPersons] = useState([]);
  const [flights, setFlights] = useState([]);
  const [stays, setStays] = useState([]);
  const [hotels, setHotels] = useState([]);
  const [pricingRules, setPricingRules] = useState([]);
  const [roommates, setRoommates] = useState([]);

  const [searchA, setSearchA] = useState(""); const [deptA, setDeptA] = useState("");
  const [searchB, setSearchB] = useState(""); const [deptB, setDeptB] = useState("");
  const [searchC, setSearchC] = useState(""); const [deptC, setDeptC] = useState("");
  const [searchD, setSearchD] = useState("");

  const [personModal, setPersonModal] = useState(null);
  const [flightModal, setFlightModal] = useState(null);
  const [stayModal, setStayModal] = useState(null);
  const [hotelModal, setHotelModal] = useState(null);
  const [priceModal, setPriceModal] = useState(null);
  const [roommateModal, setRoommateModal] = useState(null);

  const showToast = (msg) => { setToast(msg); setTimeout(() => setToast(""), 2500); };

  const loadAll = useCallback(async () => {
    setLoading(true);
    try {
      const [p, f, s, h, pr, rm] = await Promise.all([
        api.get("persons"), api.get("flights"), api.get("stays"),
        api.get("hotels"), api.get("pricing_rules"), api.get("roommates"),
      ]);
      setPersons(p); setFlights(f); setStays(s);
      setHotels(h); setPricingRules(pr); setRoommates(rm);
    } catch (e) { showToast("データ読み込みエラー: " + e.message); }
    setLoading(false);
  }, []);

  useEffect(() => { if (token) loadAll(); }, [token]);

  const handleLogin = (t, u) => { setToken(t); setUser(u); };
  const handleLogout = async () => { if (token) await api.logout(token); setToken(null); setUser(null); };

  const allDepts = useMemo(() => [...new Set(persons.map(p => p.dept).filter(Boolean))], [persons]);

  const getStatus = useCallback((pid) => {
    const f = flights.find(fl => fl.person_id === pid);
    const s = stays.find(st => st.person_id === pid);
    const hasF = f && f.airline && f.flight_no;
    const hasS = s && s.hotel_id && s.check_in && s.check_out;
    if (hasF && hasS) return "arranged";
    if (hasF || hasS) return "partial";
    return "none";
  }, [flights, stays]);

  const filterPersons = (q, dept) => persons.filter(p => {
    const kw = q.toLowerCase();
    const match = !kw || [p.name_kanji, p.last_roman, p.first_roman, p.dept, p.passport, p.diet].some(v => (v || "").toLowerCase().includes(kw));
    return match && (!dept || p.dept === dept);
  });

  const getHotelName = id => hotels.find(h => h.id == id)?.name || "—";

  // CRUD
  const savePerson = async (f) => {
    setSaving(true);
    try {
      if (personModal.mode === "add") {
        const [r] = await api.insert("persons", f);
        setPersons(p => [...p, r]);
      } else {
        const [r] = await api.update("persons", personModal.data.id, f);
        setPersons(p => p.map(x => x.id === personModal.data.id ? r : x));
      }
      showToast("✓ 保存しました"); setPersonModal(null);
    } catch (e) { showToast("エラー: " + e.message); }
    setSaving(false);
  };

  const deletePerson = async (id) => {
    if (!window.confirm("削除しますか？")) return;
    await api.delete("persons", id);
    setPersons(p => p.filter(x => x.id !== id));
    setFlights(f => f.filter(x => x.person_id !== id));
    setStays(s => s.filter(x => x.person_id !== id));
    showToast("削除しました");
  };

  const saveFlight = async (f) => {
    setSaving(true);
    try {
      const existing = flights.find(fl => fl.person_id === flightModal.pid);
      const data = { ...f, person_id: flightModal.pid };
      if (existing) {
        const [r] = await api.update("flights", existing.id, data);
        setFlights(fl => fl.map(x => x.id === existing.id ? r : x));
      } else {
        const [r] = await api.insert("flights", data);
        setFlights(fl => [...fl, r]);
      }
      showToast("✓ 保存しました"); setFlightModal(null);
    } catch (e) { showToast("エラー: " + e.message); }
    setSaving(false);
  };

  const saveStay = async (f) => {
    setSaving(true);
    try {
      const existing = stays.find(s => s.person_id === stayModal.pid);
      const data = { ...f, person_id: stayModal.pid };
      if (existing) {
        const [r] = await api.update("stays", existing.id, data);
        setStays(s => s.map(x => x.id === existing.id ? r : x));
      } else {
        const [r] = await api.insert("stays", data);
        setStays(s => [...s, r]);
      }
      showToast("✓ 保存しました"); setStayModal(null);
    } catch (e) { showToast("エラー: " + e.message); }
    setSaving(false);
  };

  const saveHotel = async (f) => {
    setSaving(true);
    try {
      if (hotelModal.mode === "add") {
        const [r] = await api.insert("hotels", f);
        setHotels(h => [...h, r]);
      } else {
        const [r] = await api.update("hotels", hotelModal.data.id, f);
        setHotels(h => h.map(x => x.id === hotelModal.data.id ? r : x));
      }
      showToast("✓ 保存しました"); setHotelModal(null);
    } catch (e) { showToast("エラー: " + e.message); }
    setSaving(false);
  };

  const deleteHotel = async (id) => {
    if (!window.confirm("削除しますか？")) return;
    await api.delete("hotels", id);
    setHotels(h => h.filter(x => x.id !== id));
    showToast("削除しました");
  };

  const savePricing = async (f) => {
    setSaving(true);
    try {
      if (priceModal.mode === "add") {
        const [r] = await api.insert("pricing_rules", f);
        setPricingRules(rs => [...rs, r]);
      } else {
        const [r] = await api.update("pricing_rules", priceModal.data.id, f);
        setPricingRules(rs => rs.map(x => x.id === priceModal.data.id ? r : x));
      }
      showToast("✓ 保存しました"); setPriceModal(null);
    } catch (e) { showToast("エラー: " + e.message); }
    setSaving(false);
  };

  const deletePricing = async (id) => {
    await api.delete("pricing_rules", id);
    setPricingRules(rs => rs.filter(r => r.id !== id));
    showToast("削除しました");
  };

  const saveRoommates = async (pid, partnerIds) => {
    setSaving(true);
    try {
      await api.deleteWhere("roommates", "person_id", pid);
      if (partnerIds.length > 0) {
        const rows = partnerIds.map(rid => ({ person_id: pid, partner_id: rid }));
        await api.insert("roommates", rows);
        // 双方向
        for (const rid of partnerIds) {
          await api.deleteWhere("roommates", "person_id", rid);
          const others = partnerIds.filter(x => x !== rid);
          const all = [...new Set([pid, ...others])];
          await api.insert("roommates", all.map(x => ({ person_id: rid, partner_id: x })));
        }
      }
      await loadAll();
      showToast("✓ 同室設定を保存しました"); setRoommateModal(null);
    } catch (e) { showToast("エラー: " + e.message); }
    setSaving(false);
  };

  const getRoommateNames = pid => roommates.filter(r => r.person_id === pid).map(r => persons.find(p => p.id === r.partner_id)?.name_kanji).filter(Boolean).join("、");

  const totalHotelCost = useMemo(() => stays.reduce((s, st) => s + (st.total_amount || 0), 0), [stays]);
  const hotelStats = useMemo(() => {
    const map = {};
    hotels.forEach(h => { map[h.id] = { guests: 0, rooms: 0, total: 0 }; });
    stays.forEach(s => {
      if (!s.hotel_id) return;
      if (!map[s.hotel_id]) map[s.hotel_id] = { guests: 0, rooms: 0, total: 0 };
      map[s.hotel_id].guests += 1; map[s.hotel_id].rooms += 1;
      map[s.hotel_id].total += (s.total_amount || 0);
    });
    return map;
  }, [stays, hotels]);

  if (!token) return <LoginScreen onLogin={handleLogin} />;

  const TABS = [
    { id: "A", label: "👤 スタッフリスト" },
    { id: "B", label: "✈ 航班管理" },
    { id: "C", label: "🏨 飯店管理" },
    { id: "D", label: "📋 飯店リスト" },
  ];

  return (
    <div style={{ fontFamily: "system-ui,sans-serif", minHeight: "100vh", background: "#f0f4f8" }}>
      {toast && (
        <div style={{ position: "fixed", top: 20, right: 20, zIndex: 999, background: "#1e3a8a", color: "white", borderRadius: 10, padding: "10px 20px", fontSize: 13, fontWeight: 600, boxShadow: "0 4px 20px rgba(0,0,0,.2)" }}>{toast}</div>
      )}
      <div style={{ background: "linear-gradient(135deg,#1e3a8a,#2563eb)", padding: "14px 24px", color: "white", display: "flex", alignItems: "center", justifyContent: "space-between" }}>
        <div>
          <div style={{ fontSize: 18, fontWeight: 800 }}>🎬 劇組交通住宿管理システム</div>
          <div style={{ fontSize: 11, opacity: .75 }}>☁ Supabase クラウド同期中 · {user?.email}</div>
        </div>
        <div style={{ display: "flex", gap: 10 }}>
          <button onClick={() => window.print()} style={{ background: "rgba(255,255,255,0.2)", color: "white", border: "1px solid rgba(255,255,255,.4)", borderRadius: 8, padding: "7px 14px", fontWeight: 600, cursor: "pointer", fontSize: 12 }}>🖨 印刷</button>
          <button onClick={handleLogout} style={{ background: "rgba(255,255,255,0.15)", color: "white", border: "1px solid rgba(255,255,255,.3)", borderRadius: 8, padding: "7px 14px", fontWeight: 600, cursor: "pointer", fontSize: 12 }}>🔒 ログアウト</button>
        </div>
      </div>

      <div className="no-print" style={{ background: "white", borderBottom: "2px solid #e5e7eb", display: "flex", paddingLeft: 14 }}>
        {TABS.map(t => (
          <button key={t.id} onClick={() => setTab(t.id)}
            style={{ padding: "11px 18px", fontWeight: tab === t.id ? 700 : 400, color: tab === t.id ? "#2563eb" : "#6b7280", borderBottom: tab === t.id ? "3px solid #2563eb" : "3px solid transparent", background: "none", border: "none", cursor: "pointer", fontSize: 13 }}>
            {t.label}
          </button>
        ))}
      </div>

      {loading && (
        <div style={{ textAlign: "center", padding: 40, color: "#6b7280", fontSize: 14 }}>☁ データを読み込んでいます…</div>
      )}

      {!loading && (
        <div style={{ padding: "18px 14px", maxWidth: 1280, margin: "0 auto" }}>

          {/* TAB A */}
          {tab === "A" && (() => {
            const rows = filterPersons(searchA, deptA);
            return (
              <div>
                <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 12, flexWrap: "wrap", gap: 8 }}>
                  <h2 style={{ fontSize: 17, fontWeight: 700, color: "#1e3a8a", margin: 0 }}>スタッフリスト</h2>
                  <div style={{ display: "flex", gap: 8, flexWrap: "wrap", alignItems: "center" }}>
                    <SearchBar value={searchA} onChange={setSearchA} placeholder="名前・部門・パスポートで検索…" />
                    <DeptFilter depts={allDepts} value={deptA} onChange={setDeptA} />
                    <button onClick={() => setPersonModal({ mode: "add", data: null })}
                      style={{ background: "#2563eb", color: "white", border: "none", borderRadius: 8, padding: "8px 16px", fontWeight: 700, cursor: "pointer", fontSize: 13 }}>＋ 追加</button>
                  </div>
                </div>
                <div style={{ overflowX: "auto" }}>
                  <table style={tableWrap}>
                    <thead><tr style={theadStyle}>{["No", "部門", "名前（漢字）", "ローマ字", "重要度", "手配状況", "パスポート", "生年月日", "有効期限", "食事", "操作"].map(h => <th key={h} style={thStyle}>{h}</th>)}</tr></thead>
                    <tbody>
                      {rows.map((p, i) => {
                        const warn = passportWarning(p.passport_exp);
                        return (
                          <tr key={p.id}>
                            <td style={tdStyle(i)}>{p.id}</td>
                            <td style={tdStyle(i)}>{p.dept}</td>
                            <td style={{ ...tdStyle(i), fontWeight: p.importance === 3 ? 700 : 400, color: p.importance === 3 ? "#dc2626" : "inherit" }}>
                              {p.name_kanji}{p.importance === 3 && <span style={{ marginLeft: 4, fontSize: 10, background: "#fee2e2", color: "#dc2626", borderRadius: 4, padding: "1px 5px" }}>★★★</span>}
                            </td>
                            <td style={tdStyle(i)}>{p.last_roman} {p.first_roman}</td>
                            <td style={{ ...tdStyle(i), color: p.importance === 3 ? "#dc2626" : p.importance === 2 ? "#d97706" : "#6b7280" }}>{starLabel(p.importance) || "—"}</td>
                            <td style={tdStyle(i)}><StatusBadge status={getStatus(p.id)} /></td>
                            <td style={tdStyle(i)}>{p.passport}</td>
                            <td style={tdStyle(i)}>{fmt(p.dob)}</td>
                            <td style={{ ...tdStyle(i), color: warn ? "#dc2626" : "inherit", fontWeight: warn ? 700 : 400 }}>{fmt(p.passport_exp)}{warn && " ⚠"}</td>
                            <td style={tdStyle(i)}>{p.diet || "—"}</td>
                            <td style={{ ...tdStyle(i), whiteSpace: "nowrap" }}>
                              <button style={editBtn} onClick={() => setPersonModal({ mode: "edit", data: p })}>編集</button>
                              <button style={delBtn} onClick={() => deletePerson(p.id)}>削除</button>
                            </td>
                          </tr>
                        );
                      })}
                      {rows.length === 0 && <tr><td colSpan={11} style={{ padding: 20, textAlign: "center", color: "#9ca3af" }}>該当なし</td></tr>}
                    </tbody>
                  </table>
                </div>
              </div>
            );
          })()}

          {/* TAB B */}
          {tab === "B" && (() => {
            const rows = filterPersons(searchB, deptB);
            return (
              <div>
                <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 12, flexWrap: "wrap", gap: 8 }}>
                  <h2 style={{ fontSize: 17, fontWeight: 700, color: "#1e3a8a", margin: 0 }}>✈ 航班管理</h2>
                  <div style={{ display: "flex", gap: 8, alignItems: "center" }}>
                    <SearchBar value={searchB} onChange={setSearchB} />
                    <DeptFilter depts={allDepts} value={deptB} onChange={setDeptB} />
                  </div>
                </div>
                <div style={{ overflowX: "auto" }}>
                  <table style={tableWrap}>
                    <thead><tr style={theadStyle}>{["No", "部門", "名前", "ローマ字", "航空会社", "便名", "Class", "PNR", "出発", "出発時刻", "到着", "到着時刻", "手荷物", "状況", "操作"].map(h => <th key={h} style={thStyle}>{h}</th>)}</tr></thead>
                    <tbody>
                      {rows.map((p, i) => {
                        const f = flights.find(fl => fl.person_id === p.id);
                        const hasF = f && f.airline;
                        return (
                          <tr key={p.id}>
                            <td style={tdStyle(i)}>{p.id}</td>
                            <td style={tdStyle(i)}>{p.dept}</td>
                            <td style={{ ...tdStyle(i), fontWeight: p.importance === 3 ? 700 : 400, color: p.importance === 3 ? "#dc2626" : "inherit" }}>{p.name_kanji}</td>
                            <td style={tdStyle(i)}>{p.last_roman} {p.first_roman}</td>
                            <td style={tdStyle(i)}>{f?.airline || "—"}</td>
                            <td style={tdStyle(i)}>{f?.flight_no || "—"}</td>
                            <td style={tdStyle(i)}>{f?.cabin || "—"}</td>
                            <td style={tdStyle(i)}>{f?.pnr || "—"}</td>
                            <td style={tdStyle(i)}>{f ? `${f.dep_airport}${f.dep_terminal ? " T" + f.dep_terminal : ""}` : "—"}</td>
                            <td style={tdStyle(i)}>{f?.dep_time ? new Date(f.dep_time).toLocaleString("ja-JP", { month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit" }) : "—"}</td>
                            <td style={tdStyle(i)}>{f ? `${f.arr_airport}${f.arr_terminal ? " T" + f.arr_terminal : ""}` : "—"}</td>
                            <td style={tdStyle(i)}>{f?.arr_time ? new Date(f.arr_time).toLocaleString("ja-JP", { month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit" }) : "—"}</td>
                            <td style={tdStyle(i)}>{f ? `預${f.checked_bag || "—"}/手${f.cabin_bag || "—"}` : "—"}</td>
                            <td style={tdStyle(i)}>{hasF ? <span style={{ padding: "2px 8px", borderRadius: 20, background: "#d1fae5", color: "#065f46", fontSize: 11, fontWeight: 700 }}>✓ 手配済</span> : <span style={{ padding: "2px 8px", borderRadius: 20, background: "#f3f4f6", color: "#9ca3af", fontSize: 11 }}>未手配</span>}</td>
                            <td style={{ ...tdStyle(i), whiteSpace: "nowrap" }}>
                              <button style={addBtn} onClick={() => setFlightModal({ pid: p.id, data: f || null })}>{hasF ? "編集" : "＋ 入力"}</button>
                            </td>
                          </tr>
                        );
                      })}
                    </tbody>
                  </table>
                </div>
              </div>
            );
          })()}

          {/* TAB C */}
          {tab === "C" && (() => {
            const rows = filterPersons(searchC, deptC);
            return (
              <div>
                <div style={{ background: "linear-gradient(90deg,#1e3a8a,#2563eb)", color: "white", borderRadius: 12, padding: "13px 20px", marginBottom: 18, display: "flex", alignItems: "center", justifyContent: "space-between" }}>
                  <div style={{ fontSize: 14, fontWeight: 600 }}>🏨 ホテル総費用（全員合計）</div>
                  <div style={{ fontSize: 24, fontWeight: 900 }}>${totalHotelCost.toLocaleString()}</div>
                </div>
                <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 12, flexWrap: "wrap", gap: 8 }}>
                  <h2 style={{ fontSize: 17, fontWeight: 700, color: "#1e3a8a", margin: 0 }}>飯店管理</h2>
                  <div style={{ display: "flex", gap: 8, alignItems: "center" }}>
                    <SearchBar value={searchC} onChange={setSearchC} />
                    <DeptFilter depts={allDepts} value={deptC} onChange={setDeptC} />
                  </div>
                </div>
                <div style={{ overflowX: "auto", marginBottom: 28 }}>
                  <table style={tableWrap}>
                    <thead><tr style={theadStyle}>{["No", "部門", "名前", "ホテル", "Room Type", "チェックイン", "チェックアウト", "泊数", "合計 ($)", "同室者", "状況", "操作"].map(h => <th key={h} style={thStyle}>{h}</th>)}</tr></thead>
                    <tbody>
                      {rows.map((p, i) => {
                        const s = stays.find(st => st.person_id === p.id);
                        const hasS = s && s.hotel_id && s.check_in;
                        const roomLabel = s?.room_type === "Custom" ? (s.room_custom || "Custom") : s?.room_type;
                        return (
                          <tr key={p.id}>
                            <td style={tdStyle(i)}>{p.id}</td>
                            <td style={tdStyle(i)}>{p.dept}</td>
                            <td style={{ ...tdStyle(i), fontWeight: p.importance === 3 ? 700 : 400, color: p.importance === 3 ? "#dc2626" : "inherit" }}>{p.name_kanji}</td>
                            <td style={tdStyle(i)}>{s ? getHotelName(s.hotel_id) : "—"}</td>
                            <td style={tdStyle(i)}>{roomLabel || "—"}</td>
                            <td style={tdStyle(i)}>{fmt(s?.check_in)}</td>
                            <td style={tdStyle(i)}>{fmt(s?.check_out)}</td>
                            <td style={tdStyle(i)}>{s?.nights || "—"}</td>
                            <td style={{ ...tdStyle(i), fontWeight: 700, color: "#2563eb" }}>{s?.total_amount ? `$${s.total_amount.toLocaleString()}` : "—"}</td>
                            <td style={tdStyle(i)}>{getRoommateNames(p.id) ? <span style={{ fontSize: 11, background: "#fef3c7", color: "#92400e", borderRadius: 4, padding: "2px 7px" }}>🛏 {getRoommateNames(p.id)}</span> : <span style={{ fontSize: 11, color: "#9ca3af" }}>個室</span>}</td>
                            <td style={tdStyle(i)}>{hasS ? <span style={{ padding: "2px 8px", borderRadius: 20, background: "#d1fae5", color: "#065f46", fontSize: 11, fontWeight: 700 }}>✓ 手配済</span> : <span style={{ padding: "2px 8px", borderRadius: 20, background: "#f3f4f6", color: "#9ca3af", fontSize: 11 }}>未手配</span>}</td>
                            <td style={{ ...tdStyle(i), whiteSpace: "nowrap" }}>
                              <button style={addBtn} onClick={() => setStayModal({ pid: p.id, data: s || null })}>{hasS ? "編集" : "＋ 入力"}</button>
                              <button style={{ ...editBtn, marginLeft: 4 }} onClick={() => setRoommateModal({ pid: p.id })}>🛏 同室</button>
                            </td>
                          </tr>
                        );
                      })}
                    </tbody>
                  </table>
                </div>
                <div>
                  <h3 style={{ fontSize: 15, fontWeight: 700, color: "#1e3a8a", marginBottom: 12 }}>📊 ホテル別統計</h3>
                  <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill,minmax(260px,1fr))", gap: 14 }}>
                    {hotels.map(h => {
                      const st = hotelStats[h.id] || { guests: 0, rooms: 0, total: 0 };
                      return (
                        <div key={h.id} style={{ background: "white", borderRadius: 12, padding: 16, boxShadow: "0 2px 8px rgba(0,0,0,.08)", borderLeft: "4px solid #2563eb" }}>
                          <div style={{ fontWeight: 700, fontSize: 14, color: "#1e3a8a", marginBottom: 12 }}>🏨 {h.name}</div>
                          <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr 1fr", gap: 8, textAlign: "center" }}>
                            <div style={{ background: "#eff6ff", borderRadius: 8, padding: "10px 6px" }}><div style={{ fontSize: 22, fontWeight: 900, color: "#2563eb" }}>{st.guests}</div><div style={{ fontSize: 10, color: "#6b7280" }}>入住人数</div></div>
                            <div style={{ background: "#f0fdf4", borderRadius: 8, padding: "10px 6px" }}><div style={{ fontSize: 22, fontWeight: 900, color: "#16a34a" }}>{st.rooms}</div><div style={{ fontSize: 10, color: "#6b7280" }}>客室数</div></div>
                            <div style={{ background: "#fefce8", borderRadius: 8, padding: "10px 6px" }}><div style={{ fontSize: 16, fontWeight: 900, color: "#ca8a04" }}>${st.total.toLocaleString()}</div><div style={{ fontSize: 10, color: "#6b7280" }}>総費用</div></div>
                          </div>
                        </div>
                      );
                    })}
                  </div>
                </div>
              </div>
            );
          })()}

          {/* TAB D */}
          {tab === "D" && (() => {
            const kw = searchD.toLowerCase();
            const filteredHotels = hotels.filter(h => !kw || [h.name, h.address, h.tel].some(v => (v || "").toLowerCase().includes(kw)));
            return (
              <div>
                <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 14, flexWrap: "wrap", gap: 8 }}>
                  <h2 style={{ fontSize: 17, fontWeight: 700, color: "#1e3a8a", margin: 0 }}>📋 飯店リスト</h2>
                  <div style={{ display: "flex", gap: 8, alignItems: "center" }}>
                    <SearchBar value={searchD} onChange={setSearchD} placeholder="ホテル名・住所・電話で検索…" />
                    <button onClick={() => setHotelModal({ mode: "add", data: null })}
                      style={{ background: "#2563eb", color: "white", border: "none", borderRadius: 8, padding: "8px 16px", fontWeight: 700, cursor: "pointer", fontSize: 13, whiteSpace: "nowrap" }}>＋ 追加</button>
                  </div>
                </div>
                <div style={{ display: "flex", flexDirection: "column", gap: 18 }}>
                  {filteredHotels.map(h => {
                    const hRules = [...pricingRules.filter(r => +r.hotel_id === h.id)].sort((a, b) => a.date > b.date ? 1 : -1);
                    return (
                      <div key={h.id} style={{ background: "white", borderRadius: 12, boxShadow: "0 2px 8px rgba(0,0,0,.08)", overflow: "hidden" }}>
                        <div style={{ borderLeft: "4px solid #2563eb", padding: "14px 16px", display: "flex", justifyContent: "space-between", alignItems: "flex-start" }}>
                          <div>
                            <div style={{ fontWeight: 700, fontSize: 15, color: "#1e3a8a", marginBottom: 4 }}>🏨 {h.name}</div>
                            <div style={{ fontSize: 12, color: "#6b7280" }}>📍 {h.address}</div>
                            <div style={{ fontSize: 12, color: "#6b7280" }}>📞 {h.tel}</div>
                          </div>
                          <div style={{ display: "flex", gap: 8, marginLeft: 12 }}>
                            <button style={editBtn} onClick={() => setHotelModal({ mode: "edit", data: h })}>編集</button>
                            <button style={delBtn} onClick={() => deleteHotel(h.id)}>削除</button>
                          </div>
                        </div>
                        <div style={{ borderTop: "1px solid #e5e7eb", padding: "12px 16px", background: "#fafafa" }}>
                          <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 8 }}>
                            <div style={{ fontSize: 13, fontWeight: 700, color: "#374151" }}>📅 日付別料金設定</div>
                            <button onClick={() => setPriceModal({ mode: "add", data: null, hotelId: h.id, hotelName: h.name })}
                              style={{ background: "#2563eb", color: "white", border: "none", borderRadius: 7, padding: "4px 12px", fontWeight: 700, cursor: "pointer", fontSize: 12 }}>＋ ルール追加</button>
                          </div>
                          {hRules.length === 0 ? <p style={{ color: "#9ca3af", fontSize: 12, margin: 0 }}>料金ルールはまだ設定されていません。</p> : (
                            <div style={{ overflowX: "auto" }}>
                              <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 12 }}>
                                <thead><tr style={{ background: "#e5e7eb" }}>{["日付", "Room Type", "基本", "重要度+", "節日+", "最終料金", "操作"].map(lbl => <th key={lbl} style={{ padding: "6px 10px", textAlign: "left", fontWeight: 600 }}>{lbl}</th>)}</tr></thead>
                                <tbody>
                                  {hRules.map(r => (
                                    <tr key={r.id} style={{ borderBottom: "1px solid #e5e7eb" }}>
                                      <td style={{ padding: "6px 10px" }}>{r.date}</td>
                                      <td style={{ padding: "6px 10px" }}>{r.room_type === "Custom" ? (r.room_custom || "Custom") : r.room_type}</td>
                                      <td style={{ padding: "6px 10px" }}>${(+r.base_price || 0).toLocaleString()}</td>
                                      <td style={{ padding: "6px 10px" }}>${(+r.importance_surcharge || 0).toLocaleString()}</td>
                                      <td style={{ padding: "6px 10px" }}>${(+r.holiday_surcharge || 0).toLocaleString()}</td>
                                      <td style={{ padding: "6px 10px", fontWeight: 700, color: "#2563eb" }}>${(+r.final_price || 0).toLocaleString()}</td>
                                      <td style={{ padding: "6px 10px", whiteSpace: "nowrap" }}>
                                        <button style={editBtn} onClick={() => setPriceModal({ mode: "edit", data: r, hotelId: h.id, hotelName: h.name })}>編集</button>
                                        <button style={delBtn} onClick={() => deletePricing(r.id)}>削除</button>
                                      </td>
                                    </tr>
                                  ))}
                                </tbody>
                              </table>
                            </div>
                          )}
                        </div>
                      </div>
                    );
                  })}
                  {filteredHotels.length === 0 && <p style={{ color: "#9ca3af", fontSize: 13 }}>該当するホテルが見つかりません</p>}
                </div>
              </div>
            );
          })()}
        </div>
      )}

      {personModal && <Modal title={personModal.mode === "add" ? "スタッフ追加" : "スタッフ編集"} onClose={() => setPersonModal(null)}><PersonForm init={personModal.data} onSave={savePerson} onClose={() => setPersonModal(null)} /></Modal>}
      {flightModal && <Modal title={`フライト情報 — ${persons.find(p => p.id === flightModal.pid)?.name_kanji || ""}`} onClose={() => setFlightModal(null)}><FlightForm init={flightModal.data} onSave={saveFlight} onClose={() => setFlightModal(null)} /></Modal>}
      {stayModal && <Modal title={`ホテル情報 — ${persons.find(p => p.id === stayModal.pid)?.name_kanji || ""}`} onClose={() => setStayModal(null)}><HotelStayForm init={stayModal.data} hotels={hotels} pricingRules={pricingRules} onSave={saveStay} onClose={() => setStayModal(null)} /></Modal>}
      {hotelModal && <Modal title={hotelModal.mode === "add" ? "ホテル追加" : "ホテル編集"} onClose={() => setHotelModal(null)}><HotelMasterForm init={hotelModal.data} onSave={saveHotel} onClose={() => setHotelModal(null)} /></Modal>}
      {priceModal && <Modal title={priceModal.mode === "add" ? "料金ルール追加" : "料金ルール編集"} onClose={() => setPriceModal(null)}><PricingRuleForm init={priceModal.data} hotelId={priceModal.hotelId} hotelName={priceModal.hotelName} onSave={savePricing} onClose={() => setPriceModal(null)} /></Modal>}
      {roommateModal && <RoommateModal pid={roommateModal.pid} persons={persons} roommates={roommates} onSave={saveRoommates} onClose={() => setRoommateModal(null)} />}

      <style>{`@media print { .no-print { display:none !important; } body { background:white; } @page { size:A4 landscape; margin:10mm; } }`}</style>
    </div>
  );
}
